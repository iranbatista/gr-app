> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Dashboard React pessoal (GitHub Pages) que consome a `gr-api` em `https://gr-api.iranbatista.dev`: lista de desafios, ranking, gráficos, feed de check-ins, página de membro e indicador de sync.

**Architecture:** SPA Vite + React + TS. `src/api/client.ts` faz fetch com `X-API-Key`; `src/api/queries.ts` expõe um hook TanStack Query por endpoint; telas em `src/routes/`, UI reutilizável em `src/components/`. `HashRouter` + `base: '/gr-app/'` para o Pages. Testes com Vitest + Testing Library + MSW renderizando o `App` inteiro dentro de `MemoryRouter`.

**Tech Stack:** Vite 8, React 19, TypeScript, TanStack Query 5, React Router 7 (`react-router-dom`), Recharts 3, Tailwind 4 (`@tailwindcss/vite`), openapi-typescript 7, Vitest 5, MSW 3, Testing Library.

**Spec:** `docs/superpowers/specs/2026-10-27-gr-app-design.md`

```ts
import { http, HttpResponse } from 'msw'
import type { Workout } from '../api/types'
import { API, challenge, oldChallenge, rankings, stats, syncMeta, workouts } from './fixtures'

export function workoutsHandler(all: Workout[]) {
  return http.get(`${API}/challenges/:id/workouts`, ({ request }) => {
    const url = new URL(request.url)
    const member = url.searchParams.get('member_id')
    const limit = Number(url.searchParams.get('limit') ?? 100)
    const offset = Number(url.searchParams.get('offset') ?? 0)
    const filtered = member ? all.filter((w) => String(w.account.id) === member) : all
    return HttpResponse.json({
      items: filtered.slice(offset, offset + limit),
      total: filtered.length,
      limit,
      offset,
    })
  })
}
```

```ts
export type Params = Record<string, string | number | null | undefined>

export class ApiError extends Error {
  status: number
  detail: string

  constructor(status: number, detail: string) {
    super(`${status}: ${detail}`)
    this.name = 'ApiError'
    this.status = status
    this.detail = detail
  }
}

// Lido em tempo de chamada (não no import) para que testes possam usar vi.stubEnv.
export function apiKey(): string {
  return import.meta.env.VITE_GYMRATS_KEY ?? ''
}

function baseUrl(): string {
  return import.meta.env.VITE_API_URL ?? ''
}

export async function apiGet<T>(path: string, params: Params = {}): Promise<T> {
  const url = new URL(path, baseUrl())
  for (const [key, value] of Object.entries(params)) {
    if (value !== undefined && value !== null && value !== '') {
      url.searchParams.set(key, String(value))
    }
  }

  let res: Response
  try {
    res = await fetch(url, { headers: { 'X-API-Key': apiKey() } })
  } catch (e) {
    throw new ApiError(0, e instanceof Error ? e.message : 'falha de rede')
  }

  if (!res.ok) {
    let detail = res.statusText
    try {
      const body = await res.json()
      if (typeof body?.detail === 'string') detail = body.detail
    } catch {
      // corpo não é JSON: mantém statusText
    }
    throw new ApiError(res.status, detail)
  }
  return (await res.json()) as T
}

export function shouldRetry(failureCount: number, error: unknown): boolean {
  if (!(error instanceof ApiError)) return false
  const transient = error.status === 0 || error.status >= 500
  return transient && failureCount < 1
}
```

```ts
import { useInfiniteQuery, useQuery } from '@tanstack/react-query'
import { apiGet } from './client'
import type { Challenge, RankingEntry, Stats, SyncMeta, WorkoutPage } from './types'

export const PAGE_SIZE = 30
const MAX_LIMIT = 1000

export interface WorkoutFilters {
  member?: number
  since?: string
  until?: string
}

// occurred_on é datetime ISO na API e a comparação é por string:
// "2026-10-20T10:00Z" > "2026-10-20", então estendemos until até o fim do dia.
export function workoutParams(f: WorkoutFilters, limit: number, offset: number) {
  return {
    limit,
    offset,
    member_id: f.member,
    since: f.since,
    until: f.until ? `${f.until}T23:59:59.999999` : undefined,
  }
}

export function nextOffset(page: WorkoutPage): number | undefined {
  const next = page.offset + page.items.length
  return next < page.total ? next : undefined
}

export function useChallenges() {
  return useQuery({
    queryKey: ['challenges'],
    queryFn: () => apiGet<Challenge[]>('/challenges'),
  })
}

export function useChallenge(id: number) {
  return useQuery({
    queryKey: ['challenge', id],
    queryFn: () => apiGet<Challenge>(`/challenges/${id}`),
  })
}

export function useRankings(id: number) {
```

```ts
const LOCALE = 'pt-BR'
const EMPTY = '—'

// "2026-10-01" seria lido como UTC meia-noite e viraria 30/09 no Brasil.
export function parseDate(iso: string): Date {
  return /^\d{4}-\d{2}-\d{2}$/.test(iso) ? new Date(`${iso}T00:00:00`) : new Date(iso)
}

const dateFmt = new Intl.DateTimeFormat(LOCALE, { day: '2-digit', month: '2-digit', year: 'numeric' })
const shortDateFmt = new Intl.DateTimeFormat(LOCALE, { day: '2-digit', month: '2-digit' })

export function formatDate(iso?: string | null): string {
  return iso ? dateFmt.format(parseDate(iso)) : EMPTY
}

export function formatShortDate(iso: string): string {
  return shortDateFmt.format(parseDate(iso))
}

export function formatPeriod(start?: string | null, end?: string | null): string {
  if (start && end) return `${formatDate(start)} – ${formatDate(end)}`
  if (start) return `desde ${formatDate(start)}`
  if (end) return `até ${formatDate(end)}`
  return EMPTY
}

export function formatDuration(minutes?: number | null): string {
  if (minutes == null) return EMPTY
  const h = Math.floor(minutes / 60)
  const m = Math.round(minutes % 60)
  if (h === 0) return `${m}min`
  return m === 0 ? `${h}h` : `${h}h ${m}min`
}

export function formatDistance(km: number): string {
  return `${km.toLocaleString(LOCALE, { minimumFractionDigits: 1, maximumFractionDigits: 1 })} km`
}

export function formatCalories(kcal: number): string {
  return `${Math.round(kcal).toLocaleString(LOCALE)} kcal`
}

export function formatPoints(points: number): string {
  return points.toLocaleString(LOCALE, { maximumFractionDigits: 1 })
}

export function formatRelative(iso?: string | null, now: Date = new Date()): string {
  if (!iso) return 'nunca'
  const minutes = Math.floor((now.getTime() - new Date(iso).getTime()) / 60_000)
  if (minutes < 1) return 'agora'
  if (minutes < 60) return `há ${minutes} min`
  const hours = Math.floor(minutes / 60)
  if (hours < 24) return `há ${hours} h`
  const days = Math.floor(hours / 24)
  return days === 1 ? 'há 1 dia' : `há ${days} dias`
}
```

```ts
import { parseDate } from './format'

export function initials(name?: string | null): string {
  const parts = (name ?? '').trim().split(/\s+/).filter(Boolean)
  if (parts.length === 0) return '?'
  const first = parts[0][0]
  const last = parts.length > 1 ? parts[parts.length - 1][0] : ''
  return (first + last).toUpperCase()
}

export function gradientFor(seed: string): string {
  let hash = 0
  for (const ch of seed) hash = (hash * 31 + ch.charCodeAt(0)) >>> 0
  const hue = hash % 360
  return `linear-gradient(135deg, hsl(${hue} 70% 50%), hsl(${(hue + 40) % 360} 70% 35%))`
}

export function isActive(c: { end_date?: string | null }, today: Date = new Date()): boolean {
  if (!c.end_date) return true
  const startOfToday = new Date(today.getFullYear(), today.getMonth(), today.getDate())
  return parseDate(c.end_date) >= startOfToday
}
```

```tsx
function message(error: unknown, notFound: string): string {
  if (!(error instanceof ApiError)) return 'Erro inesperado'
  if (error.status === 401) return 'API key inválida ou ausente no build'
  if (error.status === 404) return notFound
  if (error.status === 0) return 'Sem conexão com a API'
  return `Erro da API (${error.status})`
}

interface Props {
  error: unknown
  onRetry?: () => void
  notFound?: string
}

export function ErrorState({ error, onRetry, notFound = 'Não encontrado' }: Props) {
  const status = error instanceof ApiError ? error.status : -1
  const retryable = status === -1 || status === 0 || status >= 500
  return (
    <div role="alert" className="rounded-lg border border-red-200 bg-red-50 p-4 text-red-800">
      <p>{message(error, notFound)}</p>
      {status === 404 && (
        <Link to="/" className="mt-2 inline-block underline">
          Voltar aos desafios
        </Link>
      )}
      {retryable && onRetry && (
        <button
          type="button"
          onClick={onRetry}
          className="mt-2 rounded bg-red-600 px-3 py-1 text-white hover:bg-red-700"
        >
          Tentar de novo
```

```tsx
import { Link, useParams, useSearchParams } from 'react-router-dom'
import { useChallenge } from '../api/queries'
import { Cover } from '../components/Cover'
import { ErrorState } from '../components/ErrorState'
import { Skeleton } from '../components/Skeleton'
import { StatusBadge } from '../components/StatusBadge'
import { formatPeriod } from '../lib/format'
import { isActive } from '../lib/visual'
import { RankingTab } from './RankingTab'

const TABS = [
  { id: 'ranking', label: 'Ranking' },
  { id: 'stats', label: 'Gráficos' },
  { id: 'feed', label: 'Feed' },
] as const

type TabId = (typeof TABS)[number]['id']

function currentTab(value: string | null): TabId {
  return TABS.find((t) => t.id === value)?.id ?? 'ranking'
}

export function ChallengePage() {
  const id = Number(useParams().id)
  const [params, setParams] = useSearchParams()
  const tab = currentTab(params.get('tab'))
  const q = useChallenge(id)

  if (q.isPending) {
    return (
      <main className="mx-auto max-w-5xl space-y-4 p-4">
        <Skeleton className="h-64 w-full" />
      </main>
    )
  }
  if (q.isError) {
    return (
      <main className="mx-auto max-w-5xl p-4">
        <ErrorState error={q.error} onRetry={() => q.refetch()} notFound="Desafio não encontrado" />
      </main>
    )
  }

  const c = q.data
  return (
    <main className="mx-auto max-w-5xl space-y-4 p-4">
      <Link to="/" className="text-sm text-gray-500 hover:underline">
        ← Desafios
      </Link>
      <header className="overflow-hidden rounded-xl border bg-white">
        <Cover url={c.profile_picture_url} name={c.name} className="h-48 w-full" />
        <div className="space-y-1 p-4">

          </div>
          <p className="text-sm text-gray-500">
            {formatPeriod(c.start_date, c.end_date)} · {c.member_count ?? 0} membros
          </p>
          {c.description && <p className="text-gray-700">{c.description}</p>}
        </div>
      </header>

      <nav role="tablist" className="flex gap-1 border-b">
        {TABS.map((t) => (
          <button
            key={t.id}
            type="button"
            role="tab"
            aria-selected={tab === t.id}
            onClick={() => setParams({ tab: t.id })}
            className={`-mb-px border-b-2 px-4 py-2 ${
              tab === t.id ? 'border-orange-500 font-semibold' : 'border-transparent text-gray-500'
            }`}
          >
            {t.label}
          </button>
        ))}
      </nav>

      {tab === 'ranking' && <RankingTab challengeId={id} />}
    </main>
  )
}
```

```tsx
import { useSearchParams } from 'react-router-dom'
import { type WorkoutFilters, useRankings } from '../api/queries'
import { WorkoutList } from '../components/WorkoutList'

const inputClass = 'rounded border px-2 py-1'

export function FeedTab({ challengeId }: { challengeId: number }) {
  const [params, setParams] = useSearchParams()
  const rankings = useRankings(challengeId)

  const member = params.get('member')
  const filters: WorkoutFilters = {
    member: member ? Number(member) : undefined,
    since: params.get('since') ?? undefined,
    until: params.get('until') ?? undefined,
  }

  function update(key: string, value: string) {
    setParams((prev) => {
      const next = new URLSearchParams(prev)
      if (value) next.set(key, value)
      else next.delete(key)
      return next
    })
  }

  return (
    <div className="space-y-4">
      <div className="flex flex-wrap gap-3 rounded-lg border bg-white p-3">
        <label className="flex flex-col gap-1 text-sm">
          Membro
          <select value={member ?? ''} onChange={(e) => update('member', e.target.value)} className={inputClass}>
            <option value="">Todos</option>
            {(rankings.data ?? []).map((r) =>
              r.account.id != null ? (
                <option key={r.account.id} value={r.account.id}>
                  {r.account.full_name ?? `#${r.account.id}`}

              ) : null,

          </select>
```

```tsx
import { Avatar } from '../components/Avatar'
import { DayChart } from '../components/DayChart'
import { EmptyState } from '../components/EmptyState'
import { ErrorState } from '../components/ErrorState'
import { Skeleton } from '../components/Skeleton'
import { StatTile } from '../components/StatTile'
import { WorkoutList } from '../components/WorkoutList'
import { formatCalories, formatDistance, formatPoints } from '../lib/format'

function countByDay(items: Workout[]): DayCount[] {
  const counts = new Map<string, number>()
  for (const w of items) {
    const day = w.occurred_at?.slice(0, 10)
    if (day) counts.set(day, (counts.get(day) ?? 0) + 1)
  }
  return [...counts.entries()].sort(([a], [b]) => a.localeCompare(b)).map(([date, count]) => ({ date, count }))
}

export function MemberPage() {
  const params = useParams()
  const challengeId = Number(params.id)
  const memberId = Number(params.memberId)

  const rankings = useRankings(challengeId)
  const stats = useStats(challengeId)
  const all = useMemberWorkoutsAll(challengeId, memberId)

  const back = (
    <Link to={`/c/${challengeId}`} className="text-sm text-gray-500 hover:underline">
      ← Desafio
    </Link>
  )

  if (rankings.isPending || stats.isPending) {
    return (
      <main className="mx-auto max-w-5xl space-y-4 p-4">
        {back}
        <Skeleton className="h-24" />
        <Skeleton className="h-64" />
      </main>
    )
  }
  const failed = rankings.error ?? stats.error
  if (failed) {
    return (
      <main className="mx-auto max-w-5xl space-y-4 p-4">
        {back}
        <ErrorState
          error={failed}
          onRetry={() => {
            rankings.refetch()
            stats.refetch()
          }}
        />
      </main>
    )
  }

  const entry = rankings.data.find((r) => r.account.id === memberId)
  const totals = stats.data.by_member.find((m) => m.account_id === memberId)
  if (!entry && !totals) {
    return (
      <main className="mx-auto max-w-5xl space-y-4 p-4">
        {back}
        <EmptyState>Membro não encontrado</EmptyState>
      </main>
    )
  }

  const name = entry?.account.full_name ?? totals?.name ?? `Membro #${memberId}`

  return (
    <main className="mx-auto max-w-5xl space-y-6 p-4">
      {back}
      <header className="flex items-center gap-4">
        <Avatar url={entry?.account.profile_picture_url} name={name} size="lg" />
        <div>
          <h1 className="text-2xl font-bold">{name}</h1>
          {entry?.rank != null && <p className="text-gray-500">{entry.rank}º no ranking</p>}
        </div>
      </header>

      <div className="grid grid-cols-2 gap-3 sm:grid-cols-4">
        <StatTile label="Check-ins" value={totals?.workouts ?? 0} />
        <StatTile label="Pontos" value={formatPoints(totals?.points ?? 0)} />
        <StatTile label="Calorias" value={formatCalories(totals?.calories ?? 0)} />
        <StatTile label="Distância" value={formatDistance(totals?.distance ?? 0)} />
      </div>

      <section className="space-y-2">
        <h2 className="font-semibold">Check-ins por dia</h2>
        {all.isPending ? (
          <Skeleton className="h-64" />
        ) : all.isError ? (
          <ErrorState error={all.error} onRetry={() => all.refetch()} />
        ) : (
          <DayChart data={countByDay(all.data.items)} />
        )}
      </section>

      <section className="space-y-2">
        <h2 className="font-semibold">Histórico</h2>
        <WorkoutList challengeId={challengeId} filters={{ member: memberId }} showMember={false} />
      </section>
    </main>
  )
}
```

### Task 12: Workflow de deploy no GitHub Pages

**Files:**
- Create: `.github/workflows/deploy.yml`

- [ ] **Step 1: `.github/workflows/deploy.yml`**

```yaml
name: Deploy to GitHub Pages

on:
  push:
    branches: [main]
  workflow_dispatch:

permissions:
  contents: read
  pages: write
  id-token: write

concurrency:
  group: pages
  cancel-in-progress: true

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 24
          cache: npm
      - run: npm ci
      - run: npm test
      - run: npm run build
        env:
          VITE_API_URL: https://gr-api.iranbatista.dev
          VITE_GYMRATS_KEY: ${{ secrets.GYMRATS_KEY }}
      - uses: actions/upload-pages-artifact@v3
        with:
          path: dist

  deploy:
    needs: build
    runs-on: ubuntu-latest
    environment:
      name: github-pages
      url: ${{ steps.deployment.outputs.page_url }}
    steps:
      - id: deployment
        uses: actions/deploy-pages@v4
```
